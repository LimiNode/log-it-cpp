50 void log(const LogRecord& record, const std::string& message) override {
51 if (!record.raw_mode &&
52 static_cast<int>(record.log_level) < m_log_level.load(std::memory_order_relaxed)) {
53 return;
54 }
55
56 BufferedLogEntry entry;
57 entry.level = record.log_level;
58 entry.timestamp_ms = record.timestamp_ms;
59 entry.file = record.file;
60 entry.line = record.line;
61 entry.function = record.function;
62 entry.message = message;
63
64 LogRecordSnapshot written_snapshot;
65 {
66 std::lock_guard<std::mutex> lock(m_mutex);
67 m_evict_expired_locked(record.timestamp_ms);
68
69 // Capture an owning snapshot before moving the entry; callbacks
70 // are dispatched only after the buffer update finishes.
71 written_snapshot = to_snapshot(entry);
72 m_total_bytes += m_entry_bytes(entry);
73 m_entries.push_back(std::move(entry));
74 m_last_log_ts.store(record.timestamp_ms, std::memory_order_relaxed);
75 m_last_log_mono_ts.store(LOGIT_MONOTONIC_MS(), std::memory_order_relaxed);
76
77 m_enforce_limits_locked(record.timestamp_ms);
78 }
79
80 notify_callbacks(written_snapshot);
81 }

315 void notify_callbacks(const LogRecordSnapshot& snapshot) const {
316 std::vector<Callback> callbacks_copy;
317 {
318 std::lock_guard<std::mutex> lock(m_callbacks_mutex);
319 callbacks_copy.reserve(m_callbacks.size());
320 for (const auto& kv : m_callbacks) {
321 callbacks_copy.push_back(kv.second);
322 }
323 }
324 for (const auto& cb : callbacks_copy) {
325 try {
326 cb(snapshot);

328 }

332 // Count only the retained formatted payload, not the full object footprint.
333 static std::size_t m_entry_bytes(const BufferedLogEntry& entry) {
334 return entry.message.size();
335 }
336
337 void m_pop_front_locked() const {
338 if (m_entries.empty()) {
339 return;
340 }
341 m_total_bytes -= m_entry_bytes(m_entries.front());
342 m_entries.pop_front();
343 }
344
345 void m_evict_expired_locked(int64_t now_ms) const {
346 if (m_config.max_age_ms <= 0) {
347 return;
348 }
349
350 const int64_t cutoff = now_ms - m_config.max_age_ms;
351 while (!m_entries.empty() && m_entries.front().timestamp_ms < cutoff) {
352 m_pop_front_locked();
353 }
354 }
355
356 void m_enforce_limits_locked(int64_t now_ms) const {
357 m_evict_expired_locked(now_ms);
358
359 if (m_config.max_records > 0) {
360 while (m_entries.size() > m_config.max_records) {
361 m_pop_front_locked();
362 }
363 }
364
365 if (m_config.max_bytes > 0) {
366 // If a single message exceeds the byte budget, eviction removes it too.
367 while (m_total_bytes > m_config.max_bytes && !m_entries.empty()) {
368 m_pop_front_locked();
369 }
370 }
371 }
