120 void log(const LogRecord& record, const std::string& message) override {
121 std::unique_lock<std::mutex> lifecycle_lock(m_lifecycle_mutex);
122
123 if (m_state->stopping) {
124 ++m_dropped;
125 return;
126 }
127 m_last_log_ts = record.timestamp_ms;
128
129 OtlpLogItem item;
130 item.record = make_otlp_record_snapshot(record);
131 item.message = message;
132
133 if (!m_config.async) {
134 std::vector<OtlpLogItem> batch;
135 batch.push_back(item);
136 submit_batch_async(batch);
137 try {
138 m_client.wait_requests();
139 } catch (...) {
140 }
141 return;
142 }
143
144 for (;;) {
145 std::unique_lock<std::mutex> lock(m_state->mutex);
146 if (m_state->stopping) {
147 ++m_dropped;
148 return;
149 }
150
151 if (m_state->queue.size() >= m_config.max_queue_size) {
152 if (m_config.drop_on_overflow) {
153 ++m_dropped;
154 return;
155 }
156
157 lifecycle_lock.unlock();
158 m_state->space_cv.wait(lock, [this]() {
159 return m_state->stopping || m_state->queue.size() < m_config.max_queue_size;
160 });
161 lock.unlock();
162 lifecycle_lock.lock();
163
164 if (m_state->stopping) {
165 ++m_dropped;
166 return;
167 }
168
169 continue;
170 }
171
172 m_state->queue.push_back(item);
173 lock.unlock();
174 m_state->cv.notify_one();
175 return;
176 }
177 }

283 void worker_loop() {
284 while (true) {
285 std::vector<OtlpLogItem> batch;
286 batch.reserve(m_config.max_batch_size);
287
288 {
289 std::unique_lock<std::mutex> lock(m_state->mutex);
290 m_state->cv.wait_for(
291 lock,
292 std::chrono::milliseconds(m_config.export_interval_ms),
293 [this]() {
294 const bool can_submit =
295 !m_state->queue.empty() &&
296 m_state->http_in_flight < m_config.max_in_flight_requests;
297
298 const bool can_stop_now =
299 m_state->stopping &&
300 (m_config.cancel_on_shutdown ||
301 (m_state->queue.empty() && m_state->http_in_flight == 0));
302
303 return can_submit || can_stop_now;
304 });
305
306 if (m_state->stopping && m_config.cancel_on_shutdown) {
307 return;
308 }
309
310 if (m_state->stopping && m_state->queue.empty() && m_state->http_in_flight == 0) {
311 return;
312 }
313
314 if (m_state->queue.empty() ||
315 m_state->http_in_flight >= m_config.max_in_flight_requests) {
316 continue;
317 }
318
319 while (!m_state->queue.empty() && batch.size() < m_config.max_batch_size) {
320 batch.push_back(m_state->queue.front());
321 m_state->queue.pop_front();
322 }
323
324 m_state->space_cv.notify_all();
325 }
326
327 if (!batch.empty()) {
328 submit_batch_async(batch);
329 }
330 }

335 void submit_batch_async(const std::vector<OtlpLogItem>& batch) {
336 auto chunks = build_otlp_logs_json_payload_chunks(
337 batch, m_config.format, m_config.max_payload_bytes);
338
339 if (chunks.empty()) {
340 return;
341 }
342
343 kurlyk::Headers headers;
344 headers.emplace("Content-Type", "application/json");
345
346 auto weak_state = std::weak_ptr<OtlpHttpLoggerState>(m_state);
347
348 for (auto& chunk : chunks) {
349 std::string post_content;
350 kurlyk::Headers chunk_headers = headers;
351
352 if (m_config.compression == OtlpCompression::Gzip) {
353 if (!compress_string_gzip(chunk, post_content, m_config.compression_level)) {
354 // Compression failed: fallback to uncompressed payload.
355 // Count this as a failed export attempt so operators can observe compression issues.
356 m_state->failed_exports.fetch_add(1);
357 post_content = std::move(chunk);
358 } else {
359 chunk_headers.emplace("Content-Encoding", "gzip");
360 }
361 } else if (m_config.compression == OtlpCompression::Zstd) {
362 if (!compress_string_zstd(chunk, post_content, m_config.compression_level)) {
363 // Compression failed: fallback to uncompressed payload.
364 // Count this as a failed export attempt so operators can observe compression issues.
365 m_state->failed_exports.fetch_add(1);
366 post_content = std::move(chunk);
367 } else {
368 chunk_headers.emplace("Content-Encoding", "zstd");
369 }
370 } else {
371 post_content = std::move(chunk);
372 }
373
374 {
375 std::unique_lock<std::mutex> lock(m_state->mutex);
376 m_state->cv.wait(lock, [this]() {
377 return (m_state->stopping && m_config.cancel_on_shutdown) ||
378 m_state->http_in_flight < m_config.max_in_flight_requests;
379 });
380
381 if (m_state->stopping && m_config.cancel_on_shutdown) {
382 m_state->failed_exports.fetch_add(1);
383 continue;
384 }
385
386 ++m_state->http_in_flight;
387 }
388
389 bool submitted = false;
390
391 try {
392 submitted = m_client.post(
393 m_config.path, {}, chunk_headers, std::move(post_content),
394 [weak_state](kurlyk::HttpResponsePtr response) {
395 auto state = weak_state.lock();
396 if (!state) {
397 return;
398 }
399
400 std::lock_guard<std::mutex> lock(state->mutex);
401 if (!response || response->status_code < 200 || response->status_code >= 300) {
402 state->failed_exports.fetch_add(1);
403 }
404 if (state->http_in_flight > 0) {
405 --state->http_in_flight;
406 }
407 state->cv.notify_all();
408 });
409 } catch (...) {
410 submitted = false;
411 }
412
413 if (!submitted) {
414 auto state = m_state;
415 std::lock_guard<std::mutex> lock(state->mutex);
416 state->failed_exports.fetch_add(1);
417 if (state->http_in_flight > 0) {
418 --state->http_in_flight;
419 }
420 state->cv.notify_all();
421 }
422 }
423 }

426 void stop() {
427 {
428 std::lock_guard<std::mutex> lifecycle_lock(m_lifecycle_mutex);
429 std::lock_guard<std::mutex> lock(m_state->mutex);
430 if (m_state->stopping) {
431 return;
432 }
433 m_state->stopping = true;
434
435 if (m_config.cancel_on_shutdown) {
436 m_dropped.fetch_add(m_state->queue.size());
437 m_state->queue.clear();
438 m_state->space_cv.notify_all();
439 }
440 }
441
442 if (m_config.cancel_on_shutdown) {
443 try {
444 m_client.cancel_requests();
445 } catch (...) {
446 }
447 }
448
449 m_state->cv.notify_all();
450 m_state->space_cv.notify_all();
451
452 if (m_worker.joinable()) {
453 m_worker.join();
454 }
455
456 if (!m_config.cancel_on_shutdown) {
457 try {
458 m_client.wait_requests();
459 } catch (...) {
460 }
461 }
462 }
