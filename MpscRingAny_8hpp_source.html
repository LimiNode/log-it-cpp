86 template <class U>
87 bool try_push(U&& v) noexcept {
88 static_assert(std::is_nothrow_constructible<T, U&&>::value,
89 "MpscRingAny::try_push requires no-throw construction; pass a movable value");
90 std::size_t pos = m_enqueue_pos.load(std::memory_order_relaxed);
91 for (;;) {
92 Cell& c = m_cells[pos % m_cap];
93 std::size_t seq = c.m_seq.load(std::memory_order_acquire);
94
95 // When free, seq == pos
96 std::intptr_t diff =
97 static_cast<std::intptr_t>(seq) - static_cast<std::intptr_t>(pos);
98
99 if (diff == 0) {
100 if (m_enqueue_pos.compare_exchange_weak(
101 pos, pos + 1,
102 std::memory_order_relaxed,
103 std::memory_order_relaxed)) {
104 // We own the cell; construct T in-place.
105 new (&c.m_storage) T(std::forward<U>(v));
106 // Publish element.
107 c.m_seq.store(pos + 1, std::memory_order_release);
108 return true;
109 }
110 // CAS failed; 'pos' updated, retry.
111 } else if (diff < 0) {
112 // Full.
113 return false;
114 } else {
115 // Another producer advanced 'pos'; reload.
116 pos = m_enqueue_pos.load(std::memory_order_relaxed);
117 }
118 }
119 }

123 bool try_pop(T& out) noexcept {
124 for (;;) {
125 std::size_t pos = m_dequeue_pos.load(std::memory_order_relaxed);
126 Cell& c = m_cells[pos % m_cap];
127 std::size_t seq = c.m_seq.load(std::memory_order_acquire);
128
129 // When ready, seq == pos + 1
130 std::intptr_t diff =
131 static_cast<std::intptr_t>(seq) - static_cast<std::intptr_t>(pos + 1);
132
133 if (diff == 0) {
134 if (!m_dequeue_pos.compare_exchange_weak(
135 pos, pos + 1,
136 std::memory_order_relaxed,
137 std::memory_order_relaxed)) {
138 // Spurious failure: retry until we own the slot.
139 continue;
140 }
141
142 T* p = reinterpret_cast<T*>(&c.m_storage);
143 out = std::move(*p);
144 p->~T();
145
146 // Mark cell free for next cycle.
147 c.m_seq.store(pos + m_cap, std::memory_order_release);
148 return true;
149 }
150
151 return false; // Empty or not yet published.
152 }

169 void destroy_all_() noexcept {
170 if (!m_cells || m_cap == 0) {
171 return;
172 }
173
174 // Destruction is called only after producers and the consumer have
175 // stopped. Inspect the outstanding sequence range and destroy
176 // live objects in place; this keeps the ring usable with move-only
177 // or non-default-constructible payload types.
178 const std::size_t begin = m_dequeue_pos.load(std::memory_order_relaxed);
179 const std::size_t end = m_enqueue_pos.load(std::memory_order_relaxed);
180 for (std::size_t pos = begin; pos != end; ++pos) {
181 Cell& c = m_cells[pos % m_cap];
182 const std::size_t seq = c.m_seq.load(std::memory_order_relaxed);
183 if (seq == pos + 1) {
184 reinterpret_cast<T*>(&c.m_storage)->~T();
185 }
186 }
