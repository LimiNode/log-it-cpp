520 void handle_ansi_colors_windows(const std::string& message, std::ostream& stream) const {
521 std::string::size_type start = 0;
522 std::string::size_type pos = 0;
523
524 HANDLE handle_stdout = (&stream == &std::cerr)
525 ? GetStdHandle(STD_ERROR_HANDLE)
526 : GetStdHandle(STD_OUTPUT_HANDLE);
527
528 while ((pos = message.find("\033[", start)) != std::string::npos) {
529 // Output the part of the string before the ANSI code
530 if (pos > start) {
531 stream << message.substr(start, pos - start);
532 }
533
534 // Find the end of the ANSI code
535 std::string::size_type end_pos = message.find('m', pos);
536 if (end_pos != std::string::npos) {
537 // Extract the ANSI code
538 std::string ansi_code = message.substr(pos + 2, end_pos - pos - 2);
539 apply_color_from_ansi_code(ansi_code, handle_stdout);
540
541 // Update position
542 start = end_pos + 1;
543 } else {
544 break;
545 }
546 }
547
548 // Output any remaining part of the message
549 if (start < message.size()) {
550 stream << message.substr(start);
551 }
552 if (!message.empty()) stream << std::endl;
553
554 // Reset the console color to default
555 SetConsoleTextAttribute(handle_stdout, static_cast<WORD>(text_color_to_win_color(m_config.default_color)));
556 }

562 WORD color_value = static_cast<WORD>(text_color_to_win_color(m_config.default_color)); // Default color
563 const int code = std::stoi(ansi_code);

566 case 31: color_value = static_cast<WORD>(WinColor::DarkRed); break;
567 case 32: color_value = static_cast<WORD>(WinColor::DarkGreen); break;
568 case 33: color_value = static_cast<WORD>(WinColor::DarkYellow); break;
569 case 34: color_value = static_cast<WORD>(WinColor::DarkBlue); break;
570 case 35: color_value = static_cast<WORD>(WinColor::DarkMagenta); break;
571 case 36: color_value = static_cast<WORD>(WinColor::DarkCyan); break;
572 case 37: color_value = static_cast<WORD>(WinColor::Gray); break;
573 case 90: color_value = static_cast<WORD>(WinColor::DarkGray); break;
574 case 91: color_value = static_cast<WORD>(WinColor::Red); break;
