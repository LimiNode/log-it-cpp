32 inline std::string otlp_json_escape(const std::string& value) {
33 std::string out;
34 out.reserve(value.size() + 16);
35
36 for (size_t i = 0; i < value.size(); ++i) {
37 const unsigned char c = static_cast<unsigned char>(value[i]);
38
39 switch (c) {
40 case '"': out += "\\\""; break;
41 case '\\': out += "\\\\"; break;
42 case '\b': out += "\\b"; break;
43 case '\f': out += "\\f"; break;
44 case '\n': out += "\\n"; break;
45 case '\r': out += "\\r"; break;
46 case '\t': out += "\\t"; break;
47 default:
48 if (c < 0x20) {
49 static const char* hex = "0123456789abcdef";
50 out += "\\u00";
51 out += hex[(c >> 4) & 0x0F];
52 out += hex[c & 0x0F];
53 } else {
54 out += static_cast<char>(c);
55 }
56 break;
57 }
58 }
59
60 return out;
61 }

147 inline std::string sanitize_otlp_key(const std::string& name) {
148 std::string result = name;
149 for (char& c : result) {
150 if (!std::isalnum(static_cast<unsigned char>(c)) && c != '_' && c != '.' && c != '-') {
151 c = '_';
152 }
153 }
154 return result;

299 inline std::string build_otlp_logs_json_payload(
300 const std::vector<OtlpLogItem>& batch,
301 const OtlpJsonFormatConfig& config) {
302 std::ostringstream os;
303
304 os << '{';
305 os << "\"resourceLogs\":[{";
306 os << "\"resource\":{\"attributes\":[";
307
308 bool resource_first = true;
309 otlp_write_comma_if_needed(os, resource_first);
310 otlp_write_string_attr(os, "service.name", config.service_name);
311
312 if (!config.service_namespace.empty()) {
313 otlp_write_comma_if_needed(os, resource_first);
314 otlp_write_string_attr(os, "service.namespace", config.service_namespace);
315 }
316
317 if (!config.service_instance_id.empty()) {
318 otlp_write_comma_if_needed(os, resource_first);
319 otlp_write_string_attr(os, "service.instance.id", config.service_instance_id);
320 }
321
322 if (!config.deployment_environment.empty()) {
323 otlp_write_comma_if_needed(os, resource_first);
324 otlp_write_string_attr(os, "deployment.environment.name", config.deployment_environment);
325 }
326
327 os << "]},";
328 os << "\"scopeLogs\":[{";
329 os << "\"scope\":{\"name\":\"logit-cpp\"},";
330 os << "\"logRecords\":[";
331
332 for (size_t i = 0; i < batch.size(); ++i) {
333 if (i != 0) {
334 os << ',';
335 }
336 otlp_write_log_record_json(os, batch[i], config);
337 }
338
339 os << ']';
340 os << "}]";
341 os << "}]";
342 os << '}';
343
344 return os.str();
345 }
