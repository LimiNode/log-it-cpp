161 void log(const LogRecord& record) {
162 if (m_shutdown.load(std::memory_order_acquire)) return;
163
164 const bool targeted = record.logger_index >= 0;
165
166 std::vector<std::shared_ptr<LoggerStrategy>> snapshot;
167 snapshot.reserve(targeted ? 1 : 0);
168
169 LoggerReadLock lock(m_loggers_mx);
170 if (targeted) {
171 if (record.logger_index < static_cast<int>(m_loggers.size()))
172 snapshot.push_back(m_loggers[record.logger_index]);
173 } else {
174 snapshot = m_loggers; // copy shared_ptrs
175 }
176 lock.unlock();
177
178 if (targeted) {
179 if (snapshot.empty() || !snapshot[0]) return;
180 auto& strategy = snapshot[0];
181
182 std::lock_guard<std::mutex> exec_lock(strategy->exec_mx);
183 if (m_shutdown.load(std::memory_order_acquire)) return;
184 if (!strategy->enabled) return;
185 if (!record.raw_mode &&
186 static_cast<int>(record.log_level) < static_cast<int>(strategy->logger->get_log_level())) return;
187 dispatch_to_strategy(*strategy, record);
188 return;
189 }
190
191 for (const auto& strategy : snapshot) {
192 if (!strategy) continue;
193
194 std::lock_guard<std::mutex> exec_lock(strategy->exec_mx);
195 if (m_shutdown.load(std::memory_order_acquire)) return;
196 if (strategy->single_mode) continue;
197 if (!strategy->enabled) continue;
198 if (!record.raw_mode &&
199 static_cast<int>(record.log_level) < static_cast<int>(strategy->logger->get_log_level())) continue;
200
201 dispatch_to_strategy(*strategy, record);
202 }

406 LogClearResult clear_logger(int logger_index, const LogClearOptions& options = LogClearOptions()) {
407 LogClearResult result;
408 if (m_shutdown.load(std::memory_order_acquire)) {
409 result.ok = false;

436 LogClearResult clear_all_loggers(const LogClearOptions& options = LogClearOptions()) {
437 LogClearResult result;
438 if (m_shutdown.load(std::memory_order_acquire)) {
439 result.ok = false;
440 result.status = LogClearStatus::Failed;
441 result.message = "logger system is shut down";
442 return result;
443 }
444
445 const auto snapshot = get_all_strategy_snapshots();
446 std::size_t supported = 0;
447 std::size_t failed = 0;
448 std::size_t unsupported = 0;
449 for (const auto& strategy : snapshot) {
450 if (!strategy) continue;
451 std::lock_guard<std::mutex> exec_lock(strategy->exec_mx);
452 if (m_shutdown.load(std::memory_order_acquire)) {
453 result.ok = false;
454 result.status = LogClearStatus::Failed;
455 result.message = "logger system is shut down";
456 return result;
457 }
458 const LogClearResult one = strategy->logger->clear_logs(options);
459 if (one.status == LogClearStatus::Cleared && one.ok) {
460 ++supported;
461 result.cleared_records += one.cleared_records;
462 } else if (one.status == LogClearStatus::Unsupported) {
463 ++unsupported;
464 } else {
465 ++failed;
466 if (result.message.empty()) {
467 result.message = one.message;
468 }
469 }
470 }
471
472 result.ok = supported > 0 && failed == 0;
473 if (result.ok) {
474 result.status = LogClearStatus::Cleared;
475 result.message = unsupported > 0 ? "cleared supported loggers; skipped unsupported loggers" : "cleared";
476 } else {
477 result.status = unsupported > 0 && failed == 0 ? LogClearStatus::Unsupported : LogClearStatus::Failed;
478 if (result.message.empty()) {
479 result.message = unsupported > 0 ? "no logger supports clearing" : "no loggers to clear";
480 }
481 }
482 return result;
483 }

537 void dispatch_to_strategy(LoggerStrategy& strategy, const LogRecord& record) {
538 if (record.raw_mode) {
539 strategy.logger->log(record, record.format);
540 return;
541 }
542 if (strategy.formatter && strategy.formatter->is_passthrough()) {
543 strategy.logger->log(record, record.format);
