167 void log(const LogRecord& record) {
168 if (m_shutdown.load(std::memory_order_acquire)) return;
169
170 const bool targeted = record.logger_index >= 0;
171#ifdef LOGIT_BENCH_LEGACY_REGISTRY
172 StrategyList legacy_snapshot;
173 {
174 LoggerReadLock legacy_lock(m_loggers_mx);
175 if (targeted) {
176 if (record.logger_index < static_cast<int>(m_loggers.size())) {
177 legacy_snapshot.push_back(m_loggers[record.logger_index]);
178 }
179 } else {
180 legacy_snapshot = m_loggers;
181 }
182 }
183 const StrategyList* strategies = &legacy_snapshot;
184 // The targeted legacy snapshot contains only the selected strategy,
185 // so dispatch must address its sole element rather than the original
186 // registry index. The production snapshot retains the full registry
187 // and continues to use record.logger_index below.
188 const int strategy_index = targeted ? 0 : record.logger_index;
189#else
190 const auto snapshot = std::atomic_load_explicit(
191 &m_loggers_snapshot, std::memory_order_acquire);
192 const StrategyList* strategies = snapshot ? snapshot.get() : nullptr;
193 const int strategy_index = record.logger_index;
194#endif
195 if (!strategies) return;
196
197 if (targeted) {
198 if (strategy_index < 0 ||
199 strategy_index >= static_cast<int>(strategies->size())) return;
200 const auto& strategy = (*strategies)[strategy_index];
201 if (!strategy) return;
202
203 std::lock_guard<std::mutex> exec_lock(strategy->exec_mx);
204 if (m_shutdown.load(std::memory_order_acquire)) return;
205 if (!strategy->enabled.load(std::memory_order_relaxed)) return;
206 if (!record.raw_mode &&
207 static_cast<int>(record.log_level) < static_cast<int>(strategy->logger->get_log_level())) return;
208 dispatch_to_strategy(*strategy, record);
209 return;
210 }
211
212 for (const auto& strategy : *strategies) {
213 if (!strategy) continue;
214
215 std::lock_guard<std::mutex> exec_lock(strategy->exec_mx);
216 if (m_shutdown.load(std::memory_order_acquire)) return;
217 if (strategy->single_mode.load(std::memory_order_relaxed)) continue;
218 if (!strategy->enabled.load(std::memory_order_relaxed)) continue;
219 if (!record.raw_mode &&
220 static_cast<int>(record.log_level) < static_cast<int>(strategy->logger->get_log_level())) continue;
221
222 dispatch_to_strategy(*strategy, record);
223 }

427 LogClearResult clear_logger(int logger_index, const LogClearOptions& options = LogClearOptions()) {
428 LogClearResult result;
429 if (m_shutdown.load(std::memory_order_acquire)) {
430 result.ok = false;
431 result.status = LogClearStatus::Failed;
432 result.message = "logger system is shut down";
433 return result;
434 }
435
436 auto strategy = get_strategy_snapshot(logger_index);
437 if (!strategy) {
438 result.ok = false;
439 result.status = LogClearStatus::Failed;
440 result.message = "logger index not found";
441 return result;
442 }
443
444 std::lock_guard<std::mutex> exec_lock(strategy->exec_mx);
445 if (m_shutdown.load(std::memory_order_acquire)) {
446 result.ok = false;
447 result.status = LogClearStatus::Failed;
448 result.message = "logger system is shut down";
449 return result;

457 LogClearResult clear_all_loggers(const LogClearOptions& options = LogClearOptions()) {
458 LogClearResult result;
459 if (m_shutdown.load(std::memory_order_acquire)) {
460 result.ok = false;
461 result.status = LogClearStatus::Failed;
462 result.message = "logger system is shut down";
463 return result;
464 }
465
466 const auto snapshot = get_all_strategy_snapshots();
467 std::size_t supported = 0;
468 std::size_t failed = 0;
469 std::size_t unsupported = 0;
470 for (const auto& strategy : snapshot) {
471 if (!strategy) continue;
472 std::lock_guard<std::mutex> exec_lock(strategy->exec_mx);
473 if (m_shutdown.load(std::memory_order_acquire)) {
474 result.ok = false;
475 result.status = LogClearStatus::Failed;
476 result.message = "logger system is shut down";
477 return result;
478 }
479 const LogClearResult one = strategy->logger->clear_logs(options);
480 if (one.status == LogClearStatus::Cleared && one.ok) {
481 ++supported;
482 result.cleared_records += one.cleared_records;
483 } else if (one.status == LogClearStatus::Unsupported) {
484 ++unsupported;
485 } else {
486 ++failed;
487 if (result.message.empty()) {
488 result.message = one.message;
489 }
490 }
491 }
492
493 result.ok = supported > 0 && failed == 0;
494 if (result.ok) {
495 result.status = LogClearStatus::Cleared;
496 result.message = unsupported > 0 ? "cleared supported loggers; skipped unsupported loggers" : "cleared";
497 } else {
498 result.status = unsupported > 0 && failed == 0 ? LogClearStatus::Unsupported : LogClearStatus::Failed;
499 if (result.message.empty()) {
500 result.message = unsupported > 0 ? "no logger supports clearing" : "no loggers to clear";
501 }
502 }
503 return result;
504 }

558 void dispatch_to_strategy(LoggerStrategy& strategy, const LogRecord& record) {
559 if (record.raw_mode) {
560 strategy.logger->log(record, record.format);
561 return;
562 }
563 if (strategy.formatter && strategy.formatter->is_passthrough()) {
564 strategy.logger->log(record, record.format);
565 return;
566 }
567 const std::string msg = strategy.formatter ? strategy.formatter->format(record) : std::string();
568 strategy.logger->log(record, msg);
569 }
