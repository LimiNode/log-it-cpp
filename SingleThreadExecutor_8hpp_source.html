66 void add_task(std::function<void()> task) {
67 if (!task) return;
68 std::unique_lock<std::mutex> lock(m_mutex);
69 if (m_stop.load(std::memory_order_acquire)) return;
70 if (m_max_queue_size > 0 && m_queue.size() >= m_max_queue_size) {
71 switch (m_overflow_policy) {
72 case QueuePolicy::DropNewest:
73 ++m_dropped_tasks;
74 return;
75 case QueuePolicy::DropOldest:
76 if (!m_queue.empty()) {
77 m_queue.pop_front();
78 ++m_dropped_tasks;
79 }
80 break;
81 case QueuePolicy::Block:
82 m_cv.wait(lock, [this]() {
83 return m_max_queue_size == 0 ||
84 m_queue.size() < m_max_queue_size ||
85 m_stop.load(std::memory_order_acquire);
86 });
87 if (m_stop.load(std::memory_order_acquire)) return;
88 break;
89 }
90 }
91 m_queue.push_back(std::move(task));
92 lock.unlock();
93 m_cv.notify_one();
94 }

174 void worker_loop() {
175 for (;;) {
176 std::function<void()> task;
177 {
178 std::unique_lock<std::mutex> lock(m_mutex);
179 m_cv.wait(lock, [this]() {
180 return !m_queue.empty() || m_stop.load(std::memory_order_acquire);
181 });
182 if (m_stop.load(std::memory_order_acquire) && m_queue.empty()) {
183 break;
184 }
185 if (m_queue.empty()) continue;
186 task = std::move(m_queue.front());
187 m_queue.pop_front();
188 m_active_tasks.fetch_add(1, std::memory_order_relaxed);
189 }
190 m_cv.notify_all();
191
192 try {
193 task();
194 } catch (...) {
195 // Suppress exceptions from user tasks.
196 }
197
198 {
199 std::lock_guard<std::mutex> lock(m_mutex);
200 m_active_tasks.fetch_sub(1, std::memory_order_relaxed);
201 if (m_queue.empty() && m_active_tasks.load(std::memory_order_relaxed) == 0) {
202 m_cv.notify_all();
203 }
204 }
205 }

234 void add_task(std::function<void()> task) {
235 if (!task) return;
236 const std::shared_ptr<State> state = m_state;
237 bool schedule = false;
238
239 for (;;) {
240 bool drain_for_capacity = false;
241 {
242 std::lock_guard<std::mutex> lock(state->mutex);
243 if (state->shutdown_requested) return;
244
245 if (state->max_queue_size > 0 &&
246 state->queue.size() >= state->max_queue_size) {
247 switch (state->overflow_policy) {
248 case QueuePolicy::DropNewest:
249 ++state->dropped_tasks;
250 return;
251 case QueuePolicy::DropOldest:
252 if (!state->queue.empty()) {
253 state->queue.pop_front();
254 ++state->dropped_tasks;
255 }
256 break;
257 case QueuePolicy::Block:
258 drain_for_capacity = true;
259 break;
260 }
261 }
262
263 if (drain_for_capacity &&
264 state->max_queue_size > 0 &&
265 state->queue.size() >= state->max_queue_size) {
266 // Drain outside the lock to emulate producer backpressure
267 // without dropping the incoming task.
268 } else {
269 state->queue.push_back(std::move(task));
270 schedule = !state->scheduled;
271 state->scheduled = state->scheduled || schedule;
272 break;
273 }
274 }
275
276 drain_state(state);
277 }
278
279 if (schedule) {
280 schedule_drain(state);
281 }
282 }
