91 inline bool compress_file_gzip(const std::string& src,
92 const std::string& dst_tmp,
93 int level) {
94# if defined(LOGIT_HAS_ZLIB)
95 std::ifstream in(src.c_str(), std::ios::binary);
96 if (!in) return false;
97
98 char mode[8] = "wb6";
99 mode[2] = char('0' + clamp_level(level, 1, 9));
100 gzFile out = gzopen(dst_tmp.c_str(), mode);
101 if (!out) return false;
102 gzbuffer(out, 256 * 1024);
103
104 std::vector<char> buf(256 * 1024);
105 while (in) {
106 in.read(&buf[0], buf.size());
107 std::streamsize got = in.gcount();
108 if (got > 0) {
109 int written = gzwrite(out, &buf[0], static_cast<unsigned>(got));
110 if (written == 0) { gzclose(out); return false; }
111 }
112 }
113 return gzclose(out) == Z_OK;
114# else
115 (void)src; (void)dst_tmp; (void)level; return false;
116# endif
117 }

124 inline bool compress_file_zstd(const std::string& src,
125 const std::string& dst_tmp,
126 int level) {
127# if defined(LOGIT_HAS_ZSTD)
128 std::ifstream in(src.c_str(), std::ios::binary);
129 std::ofstream out(dst_tmp.c_str(), std::ios::binary | std::ios::trunc);
130 if (!in || !out) return false;
131
132 ZSTD_CCtx* cctx = ZSTD_createCCtx();
133 if (!cctx) return false;
134 ZSTD_CCtx_setParameter(cctx, ZSTD_c_compressionLevel, clamp_level(level, 1, 19));
135
136 const size_t inChunk = ZSTD_CStreamInSize();
137 const size_t outChunk = ZSTD_CStreamOutSize();
138 std::vector<char> inBuf(inChunk), outBuf(outChunk);
139
140 for (;;) {
141 in.read(&inBuf[0], inBuf.size());
142 size_t rd = static_cast<size_t>(in.gcount());
143 ZSTD_inBuffer zin = { &inBuf[0], rd, 0 };
144 const int last = in.eof() ? 1 : 0;
145
146 while (zin.pos < zin.size || last) {
147 ZSTD_outBuffer zout = { &outBuf[0], outBuf.size(), 0 };
148 size_t ret = last
149 ? ZSTD_compressStream2(cctx, &zout, &zin, ZSTD_e_end)
150 : ZSTD_compressStream2(cctx, &zout, &zin, ZSTD_e_continue);
151 if (ZSTD_isError(ret)) { ZSTD_freeCCtx(cctx); return false; }
152 out.write(&outBuf[0], static_cast<std::streamsize>(zout.pos));
153 if (!out) { ZSTD_freeCCtx(cctx); return false; }
154 if (last && ret == 0 && zin.pos == zin.size) {
155 ZSTD_freeCCtx(cctx);
156 out.flush();
157 return out.good();
158 }
159 if (!last && zin.pos == zin.size) break;
160 }
161 }
162# else
163 (void)src; (void)dst_tmp; (void)level; return false;
164# endif
165 }

184 inline bool compress_file(CompressType type,
185 const std::string& src,
186 int level,
187 const std::string& external_cmd) {
188 if (type == CompressType::NONE) return true;
189 if (type == CompressType::EXTERNAL_CMD) {
190 return compress_file_external(src, external_cmd, level);
191 }
192 std::string dst = src + (type == CompressType::GZIP ? ".gz" : ".zst");
193 std::string tmp = dst + ".tmp";
194 bool ok = false;
195 if (type == CompressType::GZIP) ok = compress_file_gzip(src, tmp, level);
196 else ok = compress_file_zstd(src, tmp, level);
197 if (!ok) { std::remove(tmp.c_str()); return false; }
198 if (std::rename(tmp.c_str(), dst.c_str()) != 0) { std::remove(tmp.c_str()); return false; }
199 std::remove(src.c_str());
200 return true;
201 }
