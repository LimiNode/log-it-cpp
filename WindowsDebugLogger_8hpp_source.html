267 static void write_impl(const std::string& message) {
268#if defined(_WIN32)
269 // Convert UTF-8 message to wide string for OutputDebugStringW.
270 if (message.empty()) {
271 OutputDebugStringW(L"\n");
272 return;
273 }
274 const int len = static_cast<int>(message.size());
275 const int wlen = MultiByteToWideChar(CP_UTF8, 0, message.c_str(), len, nullptr, 0);
276 if (wlen > 0) {
277 std::wstring wmsg;
278 wmsg.resize(static_cast<size_t>(wlen));
279 MultiByteToWideChar(CP_UTF8, 0, message.c_str(), len, &wmsg[0], wlen);
280 wmsg += L'\n';
281 OutputDebugStringW(wmsg.c_str());
282 } else {
283 OutputDebugStringA((message + "\n").c_str());
284 }
285#else
286 std::cerr << message << '\n';
287#endif
