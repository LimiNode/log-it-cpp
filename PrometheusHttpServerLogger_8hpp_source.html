56 explicit PrometheusHttpServerLogger(const Config& config)
57 : m_config(config) {
58
59 m_server.config.address = m_config.address;
60 m_server.config.port = m_config.port;
61
62 // Lifecycle note: this lambda captures `this`. The destructor calls stop()
63 // which invokes server.stop() and joins m_server_thread, ensuring no
64 // active handlers reference the logger after destruction.
65 m_server.resource[m_config.path]["GET"] =
66 [this](std::shared_ptr<HttpServer::Response> response,
67 std::shared_ptr<HttpServer::Request>) {
68 try {
69 m_scrapes_total.fetch_add(1);
70 m_last_scrape_timestamp_ms.store(LOGIT_CURRENT_TIMESTAMP_MS());
71 std::string payload = this->collect_payload();
72 response->write(
73 SimpleWeb::StatusCode::success_ok,
74 payload,
75 {{"Content-Type", "text/plain; version=0.0.4; charset=utf-8"},
76 {"Cache-Control", "no-store"}});
77 } catch (...) {
78 m_scrape_errors_total.fetch_add(1);
79 response->write(SimpleWeb::StatusCode::server_error_internal_server_error);
80 }
81 };
82
83 if (m_config.enable_health_endpoint) {
84 m_server.resource[m_config.health_path]["GET"] =
85 [](std::shared_ptr<HttpServer::Response> response,
86 std::shared_ptr<HttpServer::Request>) {
87 response->write(SimpleWeb::StatusCode::success_ok, "ok");
88 };
89 }
90
91 m_server.default_resource["GET"] =
92 [](std::shared_ptr<HttpServer::Response> response,
93 std::shared_ptr<HttpServer::Request>) {
94 response->write(
95 SimpleWeb::StatusCode::client_error_not_found,
96 "not found");
97 };
98
99 if (m_config.start_immediately) {
100 start();
101 }
102 }

140 std::string collect_payload() {
141 auto start = std::chrono::steady_clock::now();
142 std::vector<PrometheusMetricFamily> families;
143 {
144 std::lock_guard<std::mutex> lock(m_collect_mutex);
145 m_metrics.build_builtin_metrics(
146 families, m_config.format, "prometheus_http_server",
147 LOGIT_CURRENT_TIMESTAMP_MS());
148 build_scrape_metrics(families);
149 if (m_config.on_collect) {
150 try {
151 m_config.on_collect(families);
152 } catch (...) {
153 m_metrics.add_failed_export();
154 }
155 }
156 }
157 auto end = std::chrono::steady_clock::now();
158 m_last_collect_duration_sec.store(
159 std::chrono::duration<double>(end - start).count());
160 return build_prometheus_text_payload(families, m_config.format);
161 }
