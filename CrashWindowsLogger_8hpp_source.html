195 void write_snapshot(DWORD exception_code) const noexcept {
196 if (m_file == INVALID_HANDLE_VALUE) {
197 return;
198 }
199
200 const std::size_t capacity = m_capacity;
201 const std::size_t committed = m_committed.load(std::memory_order_acquire);
202 std::size_t available = committed;
203 if (available > capacity) {
204 available = capacity;
205 }
206 const std::size_t start_offset = committed - available;
207 const std::size_t start_index = capacity ? (start_offset % capacity) : 0;
208
209 if (available > 0) {
210 std::size_t chunk = capacity - start_index;
211 if (chunk > available) {
212 chunk = available;
213 }
214 safe_write(m_buffer.data() + start_index, chunk);
215 const std::size_t remaining = available - chunk;
216 if (remaining > 0) {
217 safe_write(m_buffer.data(), remaining);

219 }

221 write_marker(exception_code);

225 void safe_write(const char* data, std::size_t size) const noexcept {
226 if (m_file == INVALID_HANDLE_VALUE) {
227 return;
228 }
229
230 while (size > 0) {
231 DWORD to_write = static_cast<DWORD>(size);
232 DWORD written = 0;
233 if (!::WriteFile(m_file, data, to_write, &written, nullptr) || written == 0) {

250 unsigned long code = exception_code;
251 char digits[16];
252 std::size_t digit_count = 0;
253 if (code == 0) {
254 digits[digit_count++] = '0';
255 } else {
256 while (code > 0 && digit_count < sizeof(digits)) {
257 unsigned long value = code & 0xF;
258 digits[digit_count++] = static_cast<char>((value < 10) ? ('0' + value) : ('A' + (value - 10)));
259 code >>= 4;
260 }
261 }
262
263 while (digit_count > 0 && idx < sizeof(marker)) {
264 marker[idx++] = digits[--digit_count];
265 }
266
267 for (std::size_t i = 0; i < sizeof(suffix) - 1 && idx < sizeof(marker); ++i) {
268 marker[idx++] = suffix[i];
