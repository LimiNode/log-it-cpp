186 void add_task(std::function<void()> task) {
187 if (!task) return;
188# ifndef LOGIT_USE_MPSC_RING
189 std::unique_lock<std::mutex> lock(m_queue_mutex);
190 if (m_stop_flag.load(std::memory_order_acquire)) return;
191 if (m_max_queue_size > 0 && m_tasks_queue.size() >= m_max_queue_size) {
192 switch (m_overflow_policy.load(std::memory_order_relaxed)) {
193 case QueuePolicy::DropNewest:
194 ++m_dropped_tasks;
195 return;
196 case QueuePolicy::DropOldest:
197 if (!m_tasks_queue.empty()) {
198 m_tasks_queue.pop_front();
199 ++m_dropped_tasks;
200 }
201 break;
202 case QueuePolicy::Block:
203 m_queue_condition.wait(lock, [this]() {
204 return m_tasks_queue.size() < m_max_queue_size ||
205 m_stop_flag.load(std::memory_order_acquire);
206 });
207 if (m_stop_flag.load(std::memory_order_acquire)) return;
208 break;
209 }
210 }
211 m_tasks_queue.push_back(std::move(task));
212 lock.unlock();
213 m_queue_condition.notify_one();
214# else
215 enter_producer_();
216
217 std::function<void()> local_task = std::move(task);
218 bool done = false;
219
220 while (!done) {
221 if (m_stop_flag.load(std::memory_order_acquire)) {
222 break;
223 }
224
225 const auto policy = m_overflow_policy.load(std::memory_order_relaxed);
226
227 // Backpressure relies on the count of in-flight tasks.
228 if (policy == QueuePolicy::Block &&
229 m_max_queue_size > 0 &&
230 m_active_tasks.load(std::memory_order_relaxed) >= m_max_queue_size)
231 {
232 std::unique_lock<std::mutex> lk(m_cv_mutex);
233 m_cv.wait_for(lk, std::chrono::microseconds(LOGIT_TASK_EXECUTOR_BLOCK_WAIT_USEC));
234 continue;
235 }
236
237 // Try to push into the ring buffer.
238 // Move into the ring: std::function's move construction is
239 // noexcept, while copying may allocate and throw. The ring
240 // deliberately accepts only no-throw construction so a
241 // producer can never leave a claimed cell unpublished.
242 if (m_mpsc_queue.try_push(std::move(local_task))) {
243 m_cv.notify_one(); // wake the worker
244 break;
245 }
246
247 // Apply the configured overflow policy when the ring is full.
248 switch (policy) {
249 case QueuePolicy::DropNewest:
250 m_dropped_tasks.fetch_add(1, std::memory_order_relaxed);
251 done = true;
252 break;
253
254 case QueuePolicy::DropOldest:
255 // Safe MPSC behaviour: drop the incoming task.
256 // Preserves ordering and avoids producer/consumer deadlocks.
257 m_dropped_tasks.fetch_add(1, std::memory_order_relaxed);
258 done = true;
259 break;
260
261 case QueuePolicy::Block: {
262 std::unique_lock<std::mutex> lk(m_cv_mutex);
263 m_cv.wait_for(lk, std::chrono::microseconds(LOGIT_TASK_EXECUTOR_BLOCK_WAIT_USEC));
264 break;
265 }
266 }
267 }
268 leave_producer_();
269# endif
270 }

318 void set_max_queue_size(std::size_t size) {
319 std::lock_guard<std::mutex> lifecycle_lock(m_lifecycle_mutex);
320 if (m_stop_flag.load(std::memory_order_acquire)) return;
321# ifdef LOGIT_USE_MPSC_RING
322 // Tell producers to pause before any wait()/stop conditions run.
323 m_resizing.store(true, std::memory_order_release);
324 const auto deadline = std::chrono::steady_clock::now() +
325 std::chrono::seconds(1);
326 // Existing producers may be blocked by backpressure, so do not
327 // wait forever for the resize barrier.
328 if (!wait_until_producers_paused_(deadline)) {
329 m_resizing.store(false, std::memory_order_release);
330 m_resize_cv.notify_all();
331 return;
332 }
333
334 // Drain the queue completely, but do not wait forever if the worker
335 // is stalled (e.g., blocked sink or backpressure keeping
336 // m_active_tasks > 0). If we fail to drain before the deadline,
337 // abort the resize and re-open the barrier so producers can
338 // continue.
339 if (!wait_until_idle_(deadline)) {
340 m_resizing.store(false, std::memory_order_release);
341 m_resize_cv.notify_all();
342 return;
343 }
344
345 // Stop the worker so it cannot touch m_mpsc_queue during the resize.
346 std::unique_lock<std::mutex> lk(m_queue_mutex);
347 m_stop_flag.store(true, std::memory_order_relaxed);
348 lk.unlock();
349
350 m_cv.notify_all();
351 m_queue_condition.notify_all();
352 if (m_worker_thread.joinable()) {
353 m_worker_thread.join();
354 }
355
356 // Reinitialise the parameters and the ring on a single thread.
357 lk.lock();
358 m_max_queue_size = size;
359 const std::size_t cap =
360 (m_max_queue_size == 0 ? m_default_ring_cap : m_max_queue_size);
361 m_mpsc_queue = MpscRingAny<std::function<void()>>(cap);
362 // Reset counters (except drops) because the queue is empty.
363 m_active_tasks.store(0, std::memory_order_relaxed);
364 // Keep m_dropped_tasks untouched; tests manage it via macros.
365 lk.unlock();
366
367 // Clear the stop flag and restart the worker thread.
368 m_stop_flag.store(false, std::memory_order_relaxed);
369 m_worker_thread = std::thread(&TaskExecutor::worker_function, this);
370
371 // Re-open the barrier so producers can continue.
372 m_resizing.store(false, std::memory_order_release);
373 m_resize_cv.notify_all();
374# else
375 std::lock_guard<std::mutex> lock(m_queue_mutex);
376 m_max_queue_size = size;
377# endif
378 }

431 void worker_function() {
432 #ifndef LOGIT_USE_MPSC_RING
433 for (;;) {
434 std::function<void()> task;
435 std::unique_lock<std::mutex> lock(m_queue_mutex);
436 m_queue_condition.wait(lock, [this]() {
437 return !m_tasks_queue.empty() || m_stop_flag.load(std::memory_order_acquire);
438 });
439 if (m_stop_flag.load(std::memory_order_acquire) && m_tasks_queue.empty()) {
440 break;
441 }
442 task = std::move(m_tasks_queue.front());
443 m_tasks_queue.pop_front();
444 m_active_tasks.fetch_add(1, std::memory_order_relaxed);
445 lock.unlock();
446 m_queue_condition.notify_one();
447
448 task();
449
450 lock.lock();
451 m_active_tasks.fetch_sub(1, std::memory_order_relaxed);
452 if (m_tasks_queue.empty() && m_active_tasks.load(std::memory_order_relaxed) == 0) {
453 m_queue_condition.notify_all();
454 }
455 lock.unlock();
456 }
457 #else
458 for (;;) {
459 bool drained_any = false;
460 std::function<void()> task;
461
462 int budget = LOGIT_TASK_EXECUTOR_DRAIN_BUDGET;
463 while (budget--) {
464 // Count the pop attempt as active so wait() cannot observe
465 // an empty ring between try_pop() freeing a cell and the
466 // dequeued task starting execution.
467 m_active_tasks.fetch_add(1, std::memory_order_relaxed);
468 if (!m_mpsc_queue.try_pop(task)) {
469 m_active_tasks.fetch_sub(1, std::memory_order_relaxed);
470 break;
471 }
472
473 drained_any = true;
474
475 task();
476
477 m_active_tasks.fetch_sub(1, std::memory_order_relaxed);
478 m_cv.notify_one(); // freed an in-flight slot
479 }
480
481 if (queue_empty_() && m_active_tasks.load(std::memory_order_relaxed) == 0) {
482 std::unique_lock<std::mutex> lock(m_queue_mutex);
483 m_queue_condition.notify_all(); // notify wait()
484 m_cv.notify_all(); // wake producers blocked on Block
485 if (m_stop_flag.load(std::memory_order_acquire)) {
486 break;
487 }
488 }
489
490 if (!drained_any) {
491 std::unique_lock<std::mutex> lk(m_cv_mutex);
492 if (m_stop_flag.load(std::memory_order_acquire) && queue_empty_()) {
493 break;
494 }
495 m_cv.wait_for(lk, std::chrono::milliseconds(1));
496 }
497 }
498 #endif
499 }
