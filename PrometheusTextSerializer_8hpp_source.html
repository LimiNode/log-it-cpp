138 inline void prometheus_write_metric_family(
139 std::ostringstream& os,
140 const PrometheusMetricFamily& family,
141 const PrometheusTextFormatConfig& config) {
142 const std::string safe_name = prometheus_sanitize_metric_name(family.name);
143
144 if (config.include_help && !family.help.empty()) {
145 os << "# HELP " << safe_name << " " << prometheus_escape_help(family.help) << "\n";
146 }
147
148 if (config.include_type) {
149 const char* type_str = "untyped";
150 switch (family.type) {
151 case PrometheusMetricType::Counter: type_str = "counter"; break;
152 case PrometheusMetricType::Gauge: type_str = "gauge"; break;
153 default: break;
154 }
155 os << "# TYPE " << safe_name << " " << type_str << "\n";
156 }
157
158 for (size_t si = 0; si < family.samples.size(); ++si) {
159 const PrometheusSample& sample = family.samples[si];
160 const std::string sample_name = prometheus_sanitize_metric_name(sample.name);
161
162 os << sample_name;
163
164 // Merge labels: const_labels first, then sample labels, then logger/instance.
165 std::vector<PrometheusLabel> merged;
166 if (!config.const_labels.empty()) {
167 merged.insert(merged.end(), config.const_labels.begin(), config.const_labels.end());
168 }
169 if (!sample.labels.empty()) {
170 merged.insert(merged.end(), sample.labels.begin(), sample.labels.end());
171 }
172
173 if (!merged.empty()) {
174 os << "{";
175 for (size_t li = 0; li < merged.size(); ++li) {
176 if (li != 0) {
177 os << ",";
178 }
179 os << prometheus_sanitize_label_name(merged[li].name)
180 << "=\"" << prometheus_escape_label_value(merged[li].value) << "\"";
181 }
182 os << "}";
183 }
184
185 os << " ";
186 prometheus_format_value(os, sample.value);
187
188 if (config.include_timestamp && sample.timestamp_ms != 0) {
189 os << " " << sample.timestamp_ms;
190 }
191
192 os << "\n";
193 }
