904 void worker_loop() {
905 while (true) {
906 std::vector<MdbxLogItem> batch;
907 batch.reserve(m_config.max_batch_size);
908
909 {
910 std::unique_lock<std::mutex> lock(m_mutex);
911 m_cv.wait_for(
912 lock,
913 std::chrono::milliseconds(m_config.flush_interval_ms),
914 [this]() { return m_stopping || !m_queue.empty(); });
915
916 if (m_stopping && m_queue.empty()) {
917 m_idle = true;
918 m_cv.notify_all();
919 return;
920 }
921
922 if (m_queue.empty()) {
923 continue;
924 }
925
926 m_idle = false;
927 while (!m_queue.empty() && batch.size() < m_config.max_batch_size) {
928 batch.push_back(std::move(m_queue.front()));
929 m_queue.pop_front();
930 }
931 m_space_cv.notify_all();
932 }
933
934 write_batch(batch);
935
936 {
937 std::lock_guard<std::mutex> lock(m_mutex);
938 m_idle = true;
939 }
940 m_cv.notify_all();
941 }
942 }

944 void write_batch(const std::vector<MdbxLogItem>& batch) {
945 if (batch.empty()) {
946 return;
947 }
948
949 std::vector<LogRecordSnapshot> written_snapshots;
950 try {
951 std::lock_guard<std::mutex> db_lock(m_db_mutex);
952 auto txn = m_connection->transaction(mdbxc::TransactionMode::WRITABLE);
953 for (size_t i = 0; i < batch.size(); ++i) {
954 Record record = write_item_locked(batch[i], txn);
955 written_snapshots.push_back(to_log_record_snapshot(record));
956 }
957 txn.commit();
958 } catch (const std::exception& e) {
959 m_failed_writes.fetch_add(1, std::memory_order_acq_rel);
960 if (m_config.on_error) {
961 m_config.on_error(std::string("MdbxLogger write error: ") + e.what());
962 }
963 return;
964 } catch (...) {
965 m_failed_writes.fetch_add(1, std::memory_order_acq_rel);
966 if (m_config.on_error) {
967 m_config.on_error("MdbxLogger write error");
968 }
969 return;
970 }
971
972 notify_callbacks(written_snapshots);
973 }

975 Record write_item_locked(const MdbxLogItem& item, mdbxc::Transaction& txn) {
976 Record record;
977 record.session_id = m_session_id;
978 record.timestamp_ms = item.timestamp_ms;
979 record.level = item.level;
980 record.file = item.file;
981 record.function = item.function;
982 record.line = item.line;
983 record.message = item.message;
984
985 if (should_spill_payload(item.message)) {
986 Payload payload;
987 payload.payload_id = allocate_payload_id_locked(txn);
988 fill_payload(item.message, payload);
989 m_payloads->insert(payload.payload_id, payload, txn);
990 record.payload_id = payload.payload_id;
991 record.message = make_payload_preview(item.message);
992 }
993
994 for (;;) {
995 const std::string key = next_record_key_locked(record.timestamp_ms, record.sequence, txn);
996 if (m_records->insert(key, record, txn)) {
997 break;
998 }
999 advance_sequence_after_collision(record.timestamp_ms, record.sequence);
1000 }
1001 return record;
1002 }

1048 std::string next_record_key_locked(
1049 int64_t timestamp_ms,
1050 uint32_t& sequence,
1051 mdbxc::Transaction& txn) {
1052 std::unordered_map<int64_t, uint32_t>::iterator it =
1053 m_next_sequence_by_timestamp.find(timestamp_ms);
1054 uint32_t candidate = it == m_next_sequence_by_timestamp.end() ? 0 : it->second;
1055
1056 for (;;) {
1057 const std::string key = detail::make_mdbx_record_key(timestamp_ms, candidate);
1058 if (!m_records->contains(key, txn)) {
1059 sequence = candidate;
1060 if (candidate != (std::numeric_limits<uint32_t>::max)()) {
1061 m_next_sequence_by_timestamp[timestamp_ms] = candidate + 1;
1062 } else {
1063 m_next_sequence_by_timestamp[timestamp_ms] = candidate;
1064 }
1065 return key;
1066 }
1067 if (candidate == (std::numeric_limits<uint32_t>::max)()) {
1068 throw std::runtime_error("MdbxLogger: sequence exhausted for timestamp");
1069 }
1070 ++candidate;
1071 }
1072 }

1077 }

1081 static uint64_t make_unique_id() {
1082 static std::atomic<uint64_t> counter(0);
1083 const uint64_t now = static_cast<uint64_t>(LOGIT_CURRENT_TIMESTAMP_MS());
1084 const uint64_t n = counter.fetch_add(1, std::memory_order_relaxed) & 0xFFFFu;
1085 const uint64_t id = (now << 16) ^ (detail::current_process_id() << 1) ^ n;
1086 return id == 0 ? (n + 1) : id;
1087 }
