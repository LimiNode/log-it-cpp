716 void sort_log_files_newest_first(std::vector<LogFileInfo>& files) const {
717 std::sort(files.begin(), files.end(), [this](const LogFileInfo& lhs, const LogFileInfo& rhs) {
718 if (lhs.day_start_ms != rhs.day_start_ms) {
719 return lhs.day_start_ms > rhs.day_start_ms;
720 }
721 if (lhs.is_current != rhs.is_current) {
722 return lhs.is_current && !rhs.is_current;
723 }
724 const FileSortKey lhs_key = make_file_sort_key(lhs);
725 const FileSortKey rhs_key = make_file_sort_key(rhs);
726 if (lhs_key.kind_rank != rhs_key.kind_rank) {
727 return lhs_key.kind_rank > rhs_key.kind_rank;
728 }
729 if (lhs_key.primary != rhs_key.primary) {
730 return lhs_key.primary > rhs_key.primary;
731 }
732 if (lhs_key.secondary != rhs_key.secondary) {
733 return lhs_key.secondary > rhs_key.secondary;
734 }
735 if (lhs_key.artifact_name == rhs_key.artifact_name &&
736 lhs.is_compressed != rhs.is_compressed) {
737 return !lhs.is_compressed && rhs.is_compressed;
738 }
739 if (lhs_key.artifact_name != rhs_key.artifact_name) {
740 return lhs_key.artifact_name > rhs_key.artifact_name;
741 }
742 return normalize_path_separators(lhs.path) > normalize_path_separators(rhs.path);
743 });
744 }

753 static FileSortKey make_file_sort_key(const LogFileInfo& info) {
754 FileSortKey key;
755 key.artifact_name = strip_compression_suffix_impl(info.name);
756 const std::string& plain_name = key.artifact_name;
757
758 if (plain_name.size() > 15 && plain_name[10] == '_') {
759 key.kind_rank = 2;
760 const std::string rest = plain_name.substr(11, plain_name.size() - 15);
761 const size_t dot = rest.find('.');
762 const std::string time_part = dot == std::string::npos ? rest : rest.substr(0, dot);
763 key.primary = std::strtoll(time_part.c_str(), nullptr, 10);
764 if (dot != std::string::npos) {
765 key.secondary = std::strtoll(rest.substr(dot + 1).c_str(), nullptr, 10);
766 }
767 return key;

903# endif

917 }

936 // Extract timestamp and optional index for chronological sorting
937 auto extract = [&base](const std::string& name) {
938 int64_t ts = 0;
939 int idx = 0;
940 std::string rest = name.substr(base.size());
941 if (!rest.empty() && (rest[0] == '_' || rest[0] == '.')) {
942 rest = rest.substr(1);

944 ts = std::strtoll(rest.substr(0, dot).c_str(), nullptr, 10);
945 if (dot != std::string::npos) {
946 size_t dot2 = rest.find('.', dot + 1);
947 if (dot2 != std::string::npos) {
948 idx = std::atoi(rest.substr(dot + 1, dot2 - dot - 1).c_str());
949 }
950 }
951 }
952 return std::make_pair(ts, idx);
953 };

955 auto pa = extract(a.filename().string());
956 auto pb = extract(b.filename().string());
957 return (pa.first == pb.first) ? (pa.second < pb.second) : (pa.first < pb.first);
958 });
959 size_t to_remove = files.size() - max_files;
960 for (size_t i = 0; i < to_remove; ++i) {
961 fs::remove(files[i]);
962 }
963# else

965 std::vector<std::string> file_list = get_list_files(dir);
966 for (const auto& path : file_list) {
967 std::string name = path.substr(path.find_last_of("/\\") + 1);

969 files.emplace_back(path);
970 }
971 }
972 if (files.size() <= max_files) return;
973 // Extract timestamp and optional index for chronological sorting
974 auto extract = [&base](const std::string& name) {

985 idx = std::atoi(rest.substr(dot + 1, dot2 - dot - 1).c_str());
986 }
987 }

991 std::sort(files.begin(), files.end(), [&](const std::string& a, const std::string& b) {
992 auto pa = extract(a.substr(a.find_last_of("/\\") + 1));
993 auto pb = extract(b.substr(b.find_last_of("/\\") + 1));

1002# endif
1003 }
1004# endif
