73 void log(const LogRecord& record, const std::string& message) override {
74 if (!m_config.on_payload) {
75 return;
76 }
77
78 m_last_log_ts = record.timestamp_ms;
79
80 OtlpLogItem item;
81 item.record = make_otlp_record_snapshot(record);
82 item.message = message;
83
84 if (!m_config.async) {
85 {
86 std::lock_guard<std::mutex> lock(m_mutex);
87 if (m_stopping) {
88 ++m_dropped;
89 return;
90 }
91 }
92 std::vector<OtlpLogItem> batch;
93 batch.push_back(item);
94 auto chunks = build_otlp_logs_json_payload_chunks(
95 batch, m_config.format, m_config.max_payload_bytes);
96 if (m_config.on_payload) {
97 for (auto& chunk : chunks) {
98 try {
99 m_config.on_payload(std::move(chunk));
100 } catch (...) {
101 ++m_failed_exports;
102 }
103 }
104 }
105 return;
106 }
107
108 std::unique_lock<std::mutex> lock(m_mutex);
109 if (m_stopping) {
110 ++m_dropped;
111 return;
112 }
113
114 if (m_queue.size() >= m_config.max_queue_size) {
115 if (m_config.drop_on_overflow) {
116 ++m_dropped;
117 return;
118 }
119
120 m_space_cv.wait(lock, [this]() {
121 return m_stopping || m_queue.size() < m_config.max_queue_size;
122 });
123
124 if (m_stopping) {
125 ++m_dropped;
126 return;
127 }
128 }
129
130 m_queue.push_back(item);
131 lock.unlock();
132 m_cv.notify_one();
133 }

243 void worker_loop() {
244 while (true) {
245 std::vector<OtlpLogItem> batch;
246 batch.reserve(m_config.max_batch_size);
247
248 {
249 std::unique_lock<std::mutex> lock(m_mutex);
250 m_cv.wait_for(
251 lock,
252 std::chrono::milliseconds(m_config.export_interval_ms),
253 [this]() {
254 return m_stopping || !m_queue.empty();
255 });
256
257 if (m_stopping && m_queue.empty()) {
258 m_idle = true;
259 m_cv.notify_all();
260 return;
261 }
262
263 if (m_queue.empty()) {
264 continue;
265 }
266
267 m_idle = false;
268
269 while (!m_queue.empty() && batch.size() < m_config.max_batch_size) {
270 batch.push_back(m_queue.front());
271 m_queue.pop_front();
272 }
273
274 m_space_cv.notify_all();
275 }
276
277 if (!batch.empty()) {
278 auto chunks = build_otlp_logs_json_payload_chunks(
279 batch, m_config.format, m_config.max_payload_bytes);
280 if (m_config.on_payload) {
281 for (auto& chunk : chunks) {
282 try {
283 m_config.on_payload(std::move(chunk));
284 } catch (...) {
285 ++m_failed_exports;
286 }
287 }
288 }
289 }
290
291 {
292 std::lock_guard<std::mutex> lock(m_mutex);
293 m_idle = true;
294 }
295 m_cv.notify_all();
296 }
297 }
298
300 void stop() {
301 {
302 std::lock_guard<std::mutex> lock(m_mutex);
303 if (m_stopping) {
304 return;
305 }
306 m_stopping = true;
307 }
308
309 m_cv.notify_all();
310 m_space_cv.notify_all();
311
312 if (m_worker.joinable()) {
313 m_worker.join();
314 }
315 }
